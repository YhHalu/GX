cmake_minimum_required(VERSION 3.22)
project(gx_machinery LANGUAGES C ASM)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(STM32CUBE_F1_DIR "$ENV{HOME}/.local/opt/stm32/STM32CubeF1-v1.8.7" CACHE PATH "STM32CubeF1 1.8.7 SDK")
get_filename_component(MACHINERY "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMSIS "${STM32CUBE_F1_DIR}/Drivers/CMSIS")
set(DEVICE "${CMSIS}/Device/ST/STM32F1xx")
set(HAL "${STM32CUBE_F1_DIR}/Drivers/STM32F1xx_HAL_Driver")
set(STARTUP "${DEVICE}/Source/Templates/gcc/startup_stm32f103xb.s")
foreach(required IN ITEMS "${CMSIS}/Include/core_cm3.h" "${DEVICE}/Include/stm32f103xb.h" "${HAL}/Inc/stm32f1xx_hal.h" "${STARTUP}")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Missing STM32 SDK file: ${required}; set STM32CUBE_F1_DIR to the installed v1.8.7 SDK")
    endif()
endforeach()
file(STRINGS "${MACHINERY}/machinery.ioc" mcu_line REGEX "^Mcu.UserName=")
if(NOT mcu_line STREQUAL "Mcu.UserName=STM32F103C8Tx")
    message(FATAL_ERROR "This startup/linker configuration requires STM32F103C8Tx")
endif()
add_executable(machinery "${STARTUP}" runtime.c)
set(core_sources main gpio tim motor pid motor_command motor_uart stm32f1xx_it stm32f1xx_hal_msp system_stm32f1xx)
set(hal_sources stm32f1xx_hal stm32f1xx_hal_gpio stm32f1xx_hal_gpio_ex stm32f1xx_hal_tim stm32f1xx_hal_tim_ex stm32f1xx_hal_uart stm32f1xx_hal_dma stm32f1xx_hal_cortex stm32f1xx_hal_rcc stm32f1xx_hal_rcc_ex stm32f1xx_hal_pwr stm32f1xx_hal_flash stm32f1xx_hal_flash_ex stm32f1xx_hal_exti)
foreach(source IN LISTS core_sources)
    target_sources(machinery PRIVATE "${MACHINERY}/Core/Src/${source}.c")
endforeach()
foreach(source IN LISTS hal_sources)
    target_sources(machinery PRIVATE "${HAL}/Src/${source}.c")
endforeach()
target_compile_definitions(machinery PRIVATE STM32F103xB USE_HAL_DRIVER)
target_include_directories(machinery PRIVATE "${MACHINERY}/Core/Inc")
target_include_directories(machinery SYSTEM PRIVATE "${CMSIS}/Include" "${DEVICE}/Include" "${HAL}/Inc" "${HAL}/Inc/Legacy")
target_compile_options(machinery PRIVATE -mcpu=cortex-m3 -mthumb -ffunction-sections -fdata-sections -Wall -Wextra -Os -g3)
set_target_properties(machinery PROPERTIES SUFFIX ".elf" C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
set(linker "${CMAKE_CURRENT_SOURCE_DIR}/stm32f103c8.ld")
target_link_options(machinery PRIVATE -mcpu=cortex-m3 -mthumb --specs=nano.specs --specs=nosys.specs -nostartfiles "-T${linker}" -Wl,--gc-sections -Wl,--print-memory-usage "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/machinery.map")
set_property(TARGET machinery APPEND PROPERTY LINK_DEPENDS "${linker}")
target_link_libraries(machinery PRIVATE m)
add_custom_command(TARGET machinery POST_BUILD
    COMMAND "${CMAKE_OBJCOPY}" -O ihex "$<TARGET_FILE:machinery>" "${CMAKE_CURRENT_BINARY_DIR}/machinery.hex"
    COMMAND "${CMAKE_OBJCOPY}" -O binary "$<TARGET_FILE:machinery>" "${CMAKE_CURRENT_BINARY_DIR}/machinery.bin"
    COMMAND "${CMAKE_SIZE}" "$<TARGET_FILE:machinery>" VERBATIM)
